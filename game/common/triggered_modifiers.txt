peace_modifier = {
        trigger = {
		NOT = {
			AND = {
				any_owned_province = {
					NOT = {
						controlled_by = THIS
						controlled_by = REB
					}
				}
				war = yes
			}
			AND = {
				any_owned_province = {
					is_overseas = no
					is_coastal = yes
					is_blockaded = yes
				}
				war = yes
			}
			war_countries = {
				any_owned_province = {
					controlled_by = THIS
				}
			}
			war_countries = {
				any_owned_province = {
					is_overseas = no
					is_coastal = yes
					is_blockaded = yes
				}
			}
		}
		OR = {
			NOT = {
				war_countries = {
					any_owned_province = {
						is_overseas = no
						any_neighbor_province = {
							is_overseas = no
							owner = { tag = THIS }
						}
					}
				}
			}
			war = no
		}
        }
	war_exhaustion = -1
}
