# Tuberculosis
country_event = {
	id = 21000
	title = "EVTNAME21000"
	desc = "EVTDESC21000"
	picture = "disease"
	
	trigger = {
		NOT = {
			OR = {
				poor_strata_everyday_needs = 0.5
				health_care = good_health_care
			}
		}
	}

	mean_time_to_happen = {
		months = 700
		
		modifier = {
			factor = 0.75
			NOT = { poor_strata_life_needs = 0.9 }
		}
		modifier = {
			factor = 0.75
			NOT = { poor_strata_everyday_needs = 0.1 }
		}
		modifier = {
			factor = 1.5
			organic_chemistry = 1
		}
		modifier = {
			factor = 1.5
			health_care = trinket_health_care
		}
		modifier = {
			factor = 2.0
			health_care = low_health_care
		}
		modifier = {
			factor = 3.0
			health_care = acceptable_health_care
		}
	}

	option = {
		name = "EVTOPTA21000"
		random_state = {
			limit = { is_colonial = no }
			any_owned = { add_province_modifier = { name = tuberculosis_pandemic duration = 1825 }}
		}
	}
}
