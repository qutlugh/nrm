##################################
#        SUBSIDIARY FARMS        #
##################################
province_event = {
	id = 11078
	trigger = {
                owner = { NOT = { any_owned_province = { has_province_modifier = subsidiary_farm_cooldown }}}
		OR = {
			unemployment_by_type = { type=farmers value=0.10 }
			unemployment_by_type = { type=labourers value=0.10 }
		}
		NOT = {
			trade_goods = subsidiary_farm
			trade_goods = precious_metal
			COM = { capital_scope = { has_province_modifier = subsidiary_farm_cooldown }}
		}
	}

	mean_time_to_happen = { months = 1 }

	title = "EVTNAME11078"
	desc = "EVTDESC11078"

	option = {
		name = "Надеюсь это поможет им выжить"
		trade_goods = subsidiary_farm
		add_province_modifier = { name = subsidiary_farm_cooldown duration = 30 }
	}
}
