		capital_scope = { continent = europe }
		literacy = 0.3
		year = 1840
		NOT = {
			OR = {

			}
		}

	mean_time_to_happen = {
		months = 24
		modifier = {
			factor = 0.1
			has_global_flag = liberal_revolution_somewhere
		}
	}

	immediate = {
		add_country_modifier = { name=global_liberal_agitation duration=-1 }
		any_pop={limit={location={is_colonial=no}}ideology={value=liberal factor=0.05}consciousness=1}
		set_global_flag = liberal_revolution_somewhere
		set_country_flag = liberal_revolution_in_progress
	}

	option = {
		name = "EVTOPTA10001"
	}
}
