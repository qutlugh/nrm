country_event = {
	id = 70000
	title = "EVTNAME70000"
	desc = "EVTDESC70000"
	picture = "Slaves"
	trigger = {
		country = {
			slavery = yes_slavery
			tag = USA
		}
	}
	is_triggered_only = yes
	allow_multiple_instances = yes

	option = {
		name = "EVTOPTA70000"
		is_slave = yes
		country = {
			any_state = {
				limit = { is_slave = no }
				any_pop = { consciousness = 3 }
			}
			any_state = {
				limit = { is_slave = yes }
				any_pop = { consciousness = -3 }
			}
		}
		ai_chance = { factor = 0 }
	}

	option = {
		name = "EVTOPTB70000"
		is_slave = no
		country = {
			any_state = {
				limit = { is_slave = yes }
				any_pop = { consciousness = 3 }
			}
			any_state = {
				limit = { is_slave = no }
				any_pop = { consciousness = -3 }
			}
		}
		ai_chance = { factor = 1 }
	}
}

	news_desc_short = "EVTDESC1000_NEWS_SHORT"

	trigger = {
		NOT = { is_ideology_enabled = socialist }
		year = 1848
		capital_scope = { continent = europe }
		literacy = 0.4
	}
	mean_time_to_happen = { months = 6 }

	immediate = {
		enable_ideology = socialist
		any_pop = {
			limit = { OR = {
				pop_type = farmers
				pop_type = labourers
			}}

		}
		any_pop = {
			limit = { pop_type = craftsmen }
			ideology = { value = socialist factor = 0.50 }
		}
		farmers = { consciousness = 3 }
		labourers = { consciousness = 3 }
		craftsmen = { consciousness = 9 }

			limit = {
				civilized = yes
				OR = {
					capital_scope = { continent = europe }
					literacy = 0.3
				}
			}

		}
		set_global_flag = communist_manifest_was_published
	}

	option = {
		name = "С этой пропагандой необходимо бороться!"
	}
}

			}

			}

		months = 600
		modifier = {
			factor = 2
			NOT = { liberal = 0.05 }
		}
		modifier = {
			factor = 1.75
			liberal = 0.05
			NOT = { liberal = 0.10 }
		}
		modifier = {
			factor = 1.5
			liberal = 0.10
			NOT = { liberal = 0.15 }
		}
		modifier = {
			factor = 1.25
			liberal = 0.15
			NOT = { liberal = 0.20 }
		}
		modifier = {
			factor = 0.75
			liberal = 0.20
			NOT = { liberal = 0.25 }
		}
		modifier = {
			factor = 0.75
			liberal = 0.25
			NOT = { liberal = 0.30 }
		}

			factor = 0.75

		}
	}

        option = {
		name = "Дадим им эту методичку!"

			}

        }
